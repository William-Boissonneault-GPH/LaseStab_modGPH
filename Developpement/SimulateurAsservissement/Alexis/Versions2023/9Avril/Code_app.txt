        function saveOutput(app)
            try
                % Get simulation output from app.modelData
                if ~isfield(app.modelData, 'simOut') || isempty(app.modelData.simOut)
                    uialert(app.GUI, 'No simulation output available to save.', 'No Data');
                    return;
                end
                
                out = app.modelData.simOut;
                
                % Check logsout
                if ~isprop(out, 'logsout') || isempty(out.logsout)
                    uialert(app.GUI, 'No logged signals found in simulation output.', 'No Data');
                    return;
                end
                
                logsout = out.logsout;
        
                % Ensure logsout is not empty
                if logsout.numElements == 0
                    uialert(app.GUI, 'No signals found in logsout.', 'No Data');
                    return;
                end
        
                % Get reference time vector from first signal
                commonTime = logsout.get(17).Values.Time;
        
                % Initialize table with time column
                signalTable = table(commonTime, 'VariableNames', {'Time'});
        
                % Loop through all signals dynamically
                for i = 1:logsout.numElements
                    signal = logsout.get(i);
                    
                    % Clean the signal name for use as table variable name
                    cleanSignalName = matlab.lang.makeValidName(signal.Name);
                    
                    % Resample to common time base
                    resampledSignal = resample(signal.Values, commonTime);
        
                    % Add to table
                    signalTable.(cleanSignalName) = resampledSignal.Data;
                end
        
                % Ask user where to save the file
                [file, path] = uiputfile('signals.csv', 'Save Signal Log');
                if isequal(file, 0)
                    disp('User canceled file save.');
                    return;
                end
        
                % Save to CSV
                writetable(signalTable, fullfile(path, file));
        
                % Success message
                uialert(app.GUI, 'Signal data saved successfully!', 'Export Complete');
        
            catch ME
                % If anything goes wrong, show the error message
                uialert(app.GUI, sprintf('Failed to export signal data:\n%s', ME.message), 'Export Error');
            end
    end

    function saveParametersToJSON(app)
        
            % === File selection ===
            [fileName, filePath] = uiputfile('*.json', 'Save Parameters As');
            if isequal(fileName, 0)
                disp('❌ User canceled saving.');
                return;
            end
            fullFileName = fullfile(filePath, fileName);
        
            % === Prepare data ===
            data = struct();
        
            % === Export UI components (single numeric fields) ===
            allComponents = findall(app.PrmsGL);
            for i = 1:numel(allComponents)
                comp = allComponents(i);
                if isa(comp, 'matlab.ui.control.NumericEditField') && ~isempty(comp.Tag)
                    try
                        value = comp.Value;
                        data.(comp.Tag) = value;
                        fprintf('✅ Saved UI field: %s = %s\n', comp.Tag, mat2str(value));
                    catch err
                        warning('⚠️ Could not export UI field "%s": %s', comp.Tag, err.message);
                    end
                end
            end
        
            % === Export tunable variables (automated) ===
            if isfield(app.modelData, 'tunableVariables') && ~isempty(app.modelData.tunableVariables)
                tunableVars = app.modelData.tunableVariables;
                for i = 1:numel(tunableVars)
                    paramName = tunableVars(i).QualifiedName;
                    value = tunableVars(i).Value;
        
                    % Extract value if Simulink.Parameter
                    if isa(value, 'Simulink.Parameter')
                        value = value.Value;
                    end
        
                    % Save to struct
                    data.(paramName) = value;
                    fprintf('✅ Saved tunable variable: %s = %s\n', paramName, mat2str(value));
                end
            else
                warning('⚠️ No tunable variables found in app.modelData.');
            end
        
            % === Save to JSON (pretty printed) ===
            jsonText = jsonencode(data, 'PrettyPrint', true);
            fid = fopen(fullFileName, 'w');
            if fid == -1
                error('❌ Cannot create JSON file.');
            end
            fwrite(fid, jsonText, 'char');
            fclose(fid);
        
            disp(['✅ Parameters successfully saved to ', fullFileName]);
        
        end

        function loadParametresJSON(app)
            
                % === File selection ===
                [fileName, filePath] = uigetfile('*.json', 'Select JSON Parameter File');
                if isequal(fileName, 0)
                    disp('❌ User canceled loading.');
                    return;
                end
                fullFileName = fullfile(filePath, fileName);
            
                % === Read JSON ===
                try
                    jsonText = fileread(fullFileName);
                    data = jsondecode(jsonText);
                catch err
                    uialert(app.GUI, sprintf('Failed to read JSON file:\n%s', err.message), 'Load Error');
                    return;
                end
            
                % === Store last loaded parameters for reset function ===
                app.lastLoadedParameters = data;
            
                % === Apply to modelData.tunableVariables ===
                if isfield(app.modelData, 'tunableVariables') && ~isempty(app.modelData.tunableVariables)
                    for i = 1:numel(app.modelData.tunableVariables)
                        paramName = app.modelData.tunableVariables(i).QualifiedName;
                        if isfield(data, paramName)
                            newValue = data.(paramName);
                            originalSize = size(app.modelData.tunableVariables(i).Value);
                            % Reshape if necessary (for row/column vector consistency)
                            if isnumeric(newValue) && isvector(newValue) && ~isequal(size(newValue), originalSize)
                                newValue = reshape(newValue, originalSize);
                            end
                            app.modelData.tunableVariables(i).Value = newValue;
                            fprintf('✅ Loaded parameter: %s = %s\n', paramName, mat2str(newValue));
                        end
                    end
                else
                    warning('⚠️ No tunable variables found in app.modelData.');
                end
            
                % === Apply to UI components ===
                allComponents = findall(app.PrmsGL);
                for i = 1:numel(allComponents)
                    comp = allComponents(i);
                    if isa(comp, 'matlab.ui.control.NumericEditField') && ~isempty(comp.Tag)
                        if isfield(data, comp.Tag)
                            try
                                comp.Value = data.(comp.Tag);
                                fprintf('✅ Updated UI field: %s = %s\n', comp.Tag, mat2str(comp.Value));
                            catch err
                                warning('⚠️ Failed to update UI field "%s": %s', comp.Tag, err.message);
                            end
                        end
                    end
                end
            
                % === Refresh Simulation Input ===
                app.Helper.createSimInp();
            
                % === Confirmation ===
                uialert(app.GUI, 'Parameters loaded and applied successfully.', 'Load Complete');
            
            end
    
            function OpenManualButtonPushed(app)
                try
                    % Define the manual file path (app directory)
                    manualFile = fullfile(pwd, 'ManuelUtilisateur.txt');
            
                    % Check if the file exists
                    if isfile(manualFile)
                        % Open the manual in the system's default text editor
                        if ispc
                            winopen(manualFile); % Windows
                        elseif ismac
                            system(['open "', manualFile, '"']); % macOS
                        elseif isunix
                            system(['xdg-open "', manualFile, '"']); % Linux
                        else
                            uialert(app.GUI, 'Unsupported operating system.', 'Error');
                        end
                    else
                        % Show alert if file is missing
                        uialert(app.GUI, 'User manual not found in the app directory.', 'File Not Found');
                    end
                catch ME
                    % Show any unexpected error
                    uialert(app.GUI, sprintf('Error opening user manual:\n%s', ME.message), 'Error');
                end
            end

    end
